CREATE TABLE bus (
  bus_id int(11) NOT NULL AUTO_INCREMENT,
  bus_number varchar(20) NOT NULL,
  total_seats int(11) NOT NULL,
  admin_id int(11),
  route_id int(11),

  UNIQUE KEY bus_number (bus_number),
  KEY admin_id (admin_id),
  KEY route_id (route_id),
  CONSTRAINT bus_ibfk_1 FOREIGN KEY (admin_id) REFERENCES admin(admin_id),
  CONSTRAINT fk_route FOREIGN KEY (route_id) REFERENCES route(route_id)
);

CREATE TABLE booking (
  booking_id int(11) NOT NULL AUTO_INCREMENT,
  booking_date date NOT NULL,
  status enum('Confirmed','Cancelled','Pending') DEFAULT 'Pending',
  passenger_id int(11),
  bus_id int(11),
  route_id int(11),
  seat_id int(11),
  PRIMARY KEY (booking_id),
  KEY passenger_id (passenger_id),
  KEY bus_id (bus_id),
  KEY route_id (route_id),
  KEY seat_id (seat_id),
  CONSTRAINT booking_ibfk_1 FOREIGN KEY (passenger_id) REFERENCES passenger(passenger_id),
  CONSTRAINT booking_ibfk_2 FOREIGN KEY (bus_id) REFERENCES bus(bus_id),
  CONSTRAINT booking_ibfk_3 FOREIGN KEY (route_id) REFERENCES route(route_id),
  CONSTRAINT booking_ibfk_4 FOREIGN KEY (seat_id) REFERENCES seat(seat_id)
);

CREATE TABLE payment (
  payment_id int(11) NOT NULL AUTO_INCREMENT,
  booking_id int(11),
  payment_date date,
  amount decimal(8,2),
  payment_method varchar(30),
  payment_status enum('Paid','Unpaid') DEFAULT 'Paid',
  PRIMARY KEY (payment_id),
  UNIQUE KEY booking_id (booking_id),
  CONSTRAINT payment_ibfk_1 FOREIGN KEY (booking_id) REFERENCES booking(booking_id)
);
